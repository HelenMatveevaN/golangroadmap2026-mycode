package main

import (
	"bytes"
	"fmt"
	"os"
)

func main() {
	if len(os.Args) < 2 {
		fmt.Println("Укажите имя файла")
		return
	}
	filename := os.Args[1]

	data, err := os.ReadFile(filename)
	if err != nil {
		fmt.Printf("Ошибка: %g\n", err)
		return
	}

	//подсчет cnt /n
	lineCount := bytes.Count(data, []byte("\n"))

	fmt.Printf("%g %s\n", lineCount, filename)
}
